namespace App\Http\Controllers;

use App\Models\SubmissionAttempt;
use DreaM1\EfakturaConnector\Core\Contracts\DocumentSubmissionGatewayInterface;
use DreaM1\EfakturaConnector\Core\Contracts\ProviderDocumentValidationCheckInterface;
use DreaM1\EfakturaConnector\Core\Contracts\RecipientCapabilityCheckInterface;
use DreaM1\EfakturaConnector\Core\Contracts\UblDraftSubmissionInterface;
use DreaM1\EfakturaConnector\Core\Documents\IdempotencyKey;
use DreaM1\EfakturaConnector\Domain\EFaktura\Participant;
use Illuminate\Http\JsonResponse;

final class SubmitInvoiceController extends Controller
{
    private UblDraftSubmissionInterface $documentSubmission;
    private RecipientCapabilityCheckInterface $recipientCheck;
    private ProviderDocumentValidationCheckInterface $providerValidation;
    private DocumentSubmissionGatewayInterface $submission;

    public function __construct(
        UblDraftSubmissionInterface $documentSubmission,
        RecipientCapabilityCheckInterface $recipientCheck,
        ProviderDocumentValidationCheckInterface $providerValidation,
        DocumentSubmissionGatewayInterface $submission
    ) {
        $this->documentSubmission = $documentSubmission;
        $this->recipientCheck = $recipientCheck;
        $this->providerValidation = $providerValidation;
        $this->submission = $submission;
    }

    public function __invoke(SubmissionAttempt $attempt): JsonResponse
    {
        $sender = Participant::parse($attempt->senderParticipantId());
        $receiver = Participant::parse($attempt->receiverParticipantId());
        $payload = $this->documentSubmission->preparePayload(
            $sender,
            $receiver,
            $attempt->draft(),
            $attempt->creationDateTime()
        );
        $idempotencyKey = IdempotencyKey::fromString($attempt->idempotencyKey());

        // Tieto kontroly sú samostatné volania API; send() ich nespúšťa automaticky.
        $this->recipientCheck->check($payload);
        $this->providerValidation->check(new \DreaM1\EfakturaConnector\Core\Documents\ValidateDocumentRequest(
            $sender,
            $payload
        ));

        $result = $this->submission->send(new \DreaM1\EfakturaConnector\Core\Documents\SendDocumentRequest(
            $sender,
            $payload,
            $idempotencyKey
        ));

        $attempt->recordProviderResult($result);

        return response()->json([
            'status' => $result->status()->value(),
            'provider_document_id' => $result->providerDocumentId()->value(),
        ]);
    }
}

final class SubmitInvoiceErrorController extends Controller
{
    private UblDraftSubmissionInterface $documentSubmission;
    private RecipientCapabilityCheckInterface $recipientCheck;
    private ProviderDocumentValidationCheckInterface $providerValidation;
    private DocumentSubmissionGatewayInterface $submission;

    public function __construct(
        UblDraftSubmissionInterface $documentSubmission,
        RecipientCapabilityCheckInterface $recipientCheck,
        ProviderDocumentValidationCheckInterface $providerValidation,
        DocumentSubmissionGatewayInterface $submission
    ) {
        $this->documentSubmission = $documentSubmission;
        $this->recipientCheck = $recipientCheck;
        $this->providerValidation = $providerValidation;
        $this->submission = $submission;
    }

    public function __invoke(SubmissionAttempt $attempt): JsonResponse
    {
        $sender = Participant::parse($attempt->senderParticipantId());
        $receiver = Participant::parse($attempt->receiverParticipantId());
        $idempotencyKey = IdempotencyKey::fromString($attempt->idempotencyKey());
        $payload = $this->documentSubmission->preparePayload(
            $sender,
            $receiver,
            $attempt->draft(),
            $attempt->creationDateTime()
        );
        // Predbežné kontroly prebehnú pred odoslaním požiadavky cez SAPI.
        try {
            $this->recipientCheck->check($payload);
            $this->providerValidation->check(new ValidateDocumentRequest($sender, $payload));
        } catch (DocumentPreflightException $exception) {
            $attempt->markRejectedBeforeSend();

            return response()->json(['status' => 'zamietnute_pred_odoslanim'], 422);
        } catch (RequestDeferredException $exception) {
            $retryAt = $exception->retryAt()->unixTimestamp();
            $attempt->markDeferred($retryAt);

            return response()->json([
                'status' => 'odlozene',
                'retry_at' => $retryAt,
            ], 429);
        } catch (ProviderApiException $exception) {
            $attempt->markPreflightUnavailable($exception->rawStatusCode());
            Log::warning('Predbežná kontrola faktúry zlyhala.', [
                'attempt_id' => $attempt->id,
                'exception' => get_class($exception),
                'status_code' => $exception->rawStatusCode(),
            ]);

            return response()->json(['status' => 'predbezna_kontrola_nedostupna'], 503);
        }

        try {
            $result = $this->submission->send(
                new SendDocumentRequest($sender, $payload, $idempotencyKey)
            );

            $attempt->recordProviderResult($result);

            return response()->json([
                'status' => $result->status()->value(),
                'provider_document_id' => $result->providerDocumentId()->value(),
            ]);
        } catch (ProviderConnectionException $exception) {
            $attempt->markForReconciliation();
            Log::warning('Výsledok odoslania faktúry treba zosúladiť.', [
                'attempt_id' => $attempt->id,
                'exception' => get_class($exception),
            ]);

            return response()->json(['status' => 'potrebne_zosuladenie'], 202);
        } catch (RequestDeferredException $exception) {
            $retryAt = $exception->retryAt()->unixTimestamp();
            $attempt->markDeferred($retryAt);

            return response()->json([
                'status' => 'odlozene',
                'retry_at' => $retryAt,
            ], 429);
        } catch (InvalidProviderResponseException $exception) {
            $attempt->markForReconciliation();
            Log::warning('Poskytovateľ vrátil neplatnú odpoveď pri odoslaní faktúry.', [
                'attempt_id' => $attempt->id,
                'status_code' => $exception->rawStatusCode(),
            ]);

            return response()->json(['status' => 'potrebne_zosuladenie'], 202);
        } catch (ProviderErrorResponseException $exception) {
            $attempt->markProviderError($exception->rawStatusCode());
            Log::warning('Poskytovateľ vrátil HTTP chybu pri odoslaní faktúry.', [
                'attempt_id' => $attempt->id,
                'status_code' => $exception->rawStatusCode(),
            ]);

            return response()->json([
                'status' => 'chyba_poskytovatela',
                'provider_status' => $exception->rawStatusCode(),
            ], 502);
        }
    }
}

use DreaM1\EfakturaConnector\Core\Pagination\PageToken;
use DreaM1\EfakturaConnector\Core\Ubl\Contracts\UblDocumentReaderInterface;

final class ReceiveInboxController extends Controller
{
    private ReceivedDocumentProviderInterface $receivedDocuments;
    private UblDocumentReaderInterface $ublReader;

    public function __construct(
        ReceivedDocumentProviderInterface $receivedDocuments,
        UblDocumentReaderInterface $ublReader
    ) {
        $this->receivedDocuments = $receivedDocuments;
        $this->ublReader = $ublReader;
    }

    public function __invoke(Request $request, Tenant $tenant): JsonResponse
    {
        $this->authorize('update', $tenant);
        $validated = $request->validate([
            'page_token' => ['nullable', 'string'],
        ]);
        $receiverParticipant = Participant::parse($tenant->peppol_participant_id);

        $page = $this->receivedDocuments->getInboxPage(new \DreaM1\EfakturaConnector\Core\Documents\ListReceivedDocumentsRequest(
            $receiverParticipant,
            isset($validated['page_token'])
                ? PageToken::fromString($validated['page_token'])
                : null,
            PageSize::fromInt(50)
        ));

        foreach ($page->documents() as $metadata) {
            // Účastník patrí tejto organizácii; samotné metadáta doručenej pošty nestačia.
            if (! $metadata->receiver()->equals($receiverParticipant)) {
                throw new \UnexpectedValueException(
                    'Prijatý doklad nie je určený pre účastníka tejto doručenej pošty.'
                );
            }

            $reference = new ReceivedDocumentReference(
                $receiverParticipant,
                $metadata->documentId(),
                new DocumentRequestContext(
                    'received:' . $metadata->documentId()->value(),
                    null
                )
            );
            $payload = $this->receivedDocuments->getDocumentPayload($reference);
            $ublDocument = $this->ublReader->read($payload->xml());

            DB::transaction(static function () use (
                $receiverParticipant,
                $metadata,
                $payload,
                $ublDocument
            ): void {
                ReceivedInvoice::updateOrCreate(
                    [
                        'receiver_participant_id' => $receiverParticipant->identifier(),
                        'provider_document_id' => $metadata->documentId()->value(),
                    ],
                    [
                        'sender_participant_id' => $metadata->sender()->identifier(),
                        'document_type_id' => $metadata->documentTypeId()->value(),
                        'process_id' => $metadata->processId()->value(),
                        'payload_format' => $payload->payloadFormat(),
                        'document_xml' => $payload->xml(),
                    ]
                );

                // Ďalšie polia z $ublDocument namapujte do modelu hostiteľskej aplikácie.
            });

            // Potvrdenie odošlite až po dokončení miestnej transakcie.
            $this->receivedDocuments->acknowledge($reference);
        }

        return response()->json([
            'processed' => count($page->documents()),
            'next_page_token' => $page->nextPageToken() === null
                ? null
                : $page->nextPageToken()->value(),
        ]);
    }
}

        $attempt->draft(),

$pageToken = null;

do {
    $page = $receivedDocuments->getInboxPage(new \DreaM1\EfakturaConnector\Core\Documents\ListReceivedDocumentsRequest(
        $participant,
        $pageToken,
        PageSize::fromInt(50)
    ));

    foreach ($page->documents() as $metadata) {
        if (! $metadata->receiver()->equals($participant)) {
            throw new UnexpectedValueException('Doklad nepatrí do tejto doručenej pošty.');
        }

        $reference = new ReceivedDocumentReference(
            $participant,
            $metadata->documentId(),
            new DocumentRequestContext(
                'received:' . $metadata->documentId()->value(),

            )
        );

        // Toto úložisko patrí hostiteľskej aplikácii a musí byť idempotentné.
        $repository->storeIfAbsent(
            $participant,
            $metadata,
            $payload,
            $document
        );

        // Potvrdzujte až po úspešnom uložení.
        $receivedDocuments->acknowledge($reference);
    }

    $pageToken = $page->nextPageToken();
} while ($pageToken !== null);
